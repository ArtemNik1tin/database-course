SELECT SUM(distance_back) as total_route_distance
FROM route_by_stops rbs
WHERE rbs.route_number = 11
  AND rbs.id_direction = 1
  AND rbs.stop_number > 1
  AND rbs.stop_number <= 21;
-- Ответ: 10220

-- Задание 8
WITH route_segment AS (
    SELECT id_stop, stop_number, distance_back
    FROM route_by_stops rbs
    WHERE rbs.route_number = 11
      AND rbs.id_direction = 1
),
running_distances AS (
    SELECT r1.id_stop,
           SUM(r2.distance_back) as current_distance
    FROM route_segment r1
    JOIN route_segment r2 ON r2.stop_number > 1 AND r2.stop_number <= r1.stop_number
    WHERE r1.stop_number > 1
      AND r1.stop_number <= 36
    GROUP BY r1.id_stop
)
SELECT id_stop
FROM running_distances
ORDER BY ABS(current_distance - (17920 / 2.0)) ASC
LIMIT 1;
-- Ответ: 14981
